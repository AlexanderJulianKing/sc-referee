"""Transit detection yield among planned TESS candidate light curves.

Reads inputs/data.csv, applies the prespecified photometric screening rule,
counts confirmed transit detections among the retained curves, and writes
results/report.md. The selected rate uses the complete planned set as its
exposure denominator.
"""

import csv
from decimal import Decimal, ROUND_HALF_EVEN
from pathlib import Path

MAX_BASELINE_RMS_PPM = 1500

ROOT = Path(__file__).resolve().parent.parent
DATA_PATH = ROOT / "inputs" / "data.csv"
REPORT_PATH = ROOT / "results" / "report.md"


def load_rows(path):
    with path.open(newline="", encoding="ascii") as handle:
        return list(csv.DictReader(handle))


def is_eligible(row):
    return int(row["baseline_rms_ppm"]) <= MAX_BASELINE_RMS_PPM


def percent(numerator, denominator):
    value = Decimal(numerator) / Decimal(denominator) * Decimal(100)
    return str(value.quantize(Decimal("0.001"), rounding=ROUND_HALF_EVEN))


def main():
    rows = load_rows(DATA_PATH)
    planned = len(rows)
    retained = [row for row in rows if is_eligible(row)]
    removed = [row for row in rows if not is_eligible(row)]
    n_retained = len(retained)
    n_removed = len(removed)
    events = sum(1 for row in retained if row["transit_detected"] == "yes")
    rate = percent(events, planned)

    lines = [
        "# Transit Detection Yield in Planned TESS Candidate Light Curves",
        "",
        "## Scientific target",
        "",
        "The scientific target is the complete planned set of "
        + str(planned)
        + " candidate light curves",
        "selected for transit vetting, including the curves that the prespecified",
        "screening step removed.",
        "",
        "## Planned unit accounting",
        "",
        "- Planned candidate light curves: " + str(planned),
        "- Retained after screening: " + str(n_retained),
        "- Removed by screening: " + str(n_removed),
        "- Confirmed transit detections among retained curves: " + str(events),
        "",
        "## Screening step",
        "",
        "Prespecified before vetting: a candidate light curve is eligible if its baseline",
        "photometric scatter is at most "
        + str(MAX_BASELINE_RMS_PPM)
        + " ppm. The "
        + str(n_removed)
        + " curves above that threshold were",
        "removed and were not searched for transits.",
        "",
        "## Selected result",
        "",
        "[selected-result] Confirmed transit detections per planned candidate light curve: "
        + str(events)
        + " / "
        + str(planned)
        + " = "
        + rate
        + " percent.",
        "",
        "## Scope of the claim",
        "",
        "The selected rate uses the complete planned set of "
        + str(planned)
        + " curves as its exposure",
        "denominator, so it applies to the declared target. Because the "
        + str(n_removed)
        + " removed curves",
        "were never searched, each is counted in the denominator as a non-detection; the",
        "rate is therefore a lower bound on the detection rate that full-sensitivity",
        "vetting of all " + str(planned) + " curves would have produced. No claim is made about the",
        "detection rate within the retained subset alone.",
        "",
        "Generated by workflow/analysis.py from inputs/data.csv.",
    ]

    text = "\n".join(lines) + "\n"
    REPORT_PATH.parent.mkdir(parents=True, exist_ok=True)

import csv
from decimal import Decimal
from fractions import Fraction
from pathlib import Path

data_path = Path("inputs/data.csv")
out_dir = Path("results")
out_dir.mkdir(parents=True, exist_ok=True)

with data_path.open(newline="") as f:
    rows = list(csv.DictReader(f))

planned = len(rows)
retained = [r for r in rows if r["condition"] == "intact"]
retained_n = len(retained)
removed = planned - retained_n
occupied = sum(1 for r in retained if r["occupied"] == "yes")

rate = Fraction(occupied, retained_n)
percent = Decimal(rate.numerator * 100) / Decimal(rate.denominator)

lines = [
    "# Bat Roost Box Occupancy Survey, Willow Creek Woodland",
    "",
    "## Accounting",
    "",
    f"- Planned roost boxes: {planned}",
    f"- Retained after intact-condition screening: {retained_n}",
    f"- Removed by screening (damaged or inaccessible): {removed}",
    f"- Occupancy events among retained boxes: {occupied}",
    "",
    "## Selected result",
    "",
    (
        f"[selected-result] Among the {retained_n} retained intact roost boxes "
        f"(the prespecified scientific target; the {removed} screened-out boxes "
        f"from the {planned} planned are excluded), {occupied} showed bat occupancy, "
        f"an occupancy rate of {percent:.1f}% ({occupied}/{retained_n})."
    ),
    "",
    (
        f"All claims here apply only to the retained post-screening subset of "
        f"{retained_n} intact boxes, not to the complete planned set of {planned} boxes."
    ),
]

(out_dir / "report.md").write_text("\n".join(lines) + "\n")
